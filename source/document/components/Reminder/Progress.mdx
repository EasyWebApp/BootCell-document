import { ProgressBar } from 'boot-cell';

import { Example } from '../../../component/Example';

Use `now` with `min` and `max` to set progress. `variant` accepts semantic theme
names such as `primary`, `success`, `accent`, and `inverse`.

<Example>
    <ProgressBar now={25} label />
    <ProgressBar className="mt-3" now={60} variant="success" label />
    <ProgressBar className="mt-3" now={80} variant="accent" striped animated />
</Example>

```tsx
import { ProgressBar } from 'boot-cell';

<ProgressBar min={0} max={200} now={120} label={value => `${value} points`} />;
```

The progress indicator uses `theme-*` classes. Replace legacy color aliases
with values from BootCell's exported `ThemeColor` type.
